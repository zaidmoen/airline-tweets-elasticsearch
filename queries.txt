# Run in Kibana Dev Tools Console. All queries target the tweets index.

# Q1 - Total negative tweets and negative United tweets.
GET /tweets/_search
{
  "size": 0,
  "aggs": {
    "negative_tweets": {
      "filter": {"term": {"airline_sentiment": "negative"}},
      "aggs": {
        "united": {"filter": {"term": {"airline": "United"}}}
      }
    }
  }
}

# Q2 - Require both a baggage term and a loss or delay term.
GET /tweets/_search
{
  "size": 5,
  "track_total_hits": true,
  "_source": ["text", "airline", "airline_sentiment", "created_at"],
  "query": {
    "bool": {
      "must": [
        {"match": {"text": "luggage bags baggage suitcase"}},
        {"match": {"text": "lost delayed missing"}}
      ]
    }
  },
  "sort": [{"created_at": "desc"}, {"tweet_id": "asc"}]
}

# Q3 - Positive tweets that contain the exact phrase customer service.
GET /tweets/_search
{
  "size": 0,
  "track_total_hits": true,
  "query": {
    "bool": {
      "filter": [{"term": {"airline_sentiment": "positive"}}],
      "must": [{"match_phrase": {"text": "customer service"}}]
    }
  }
}

# Q4 - Negative Delta late-flight tweets on February 23 in Pacific time.
GET /tweets/_search
{
  "size": 100,
  "query": {
    "bool": {
      "filter": [
        {"term": {"airline_sentiment": "negative"}},
        {"term": {"airline": "Delta"}},
        {"term": {"negativereason": "Late Flight"}},
        {"range": {
          "created_at": {
            "gte": "2015-02-23T00:00:00",
            "lt": "2015-02-24T00:00:00",
            "time_zone": "America/Los_Angeles"
          }
        }}
      ]
    }
  }
}

# Q11 - Tweets within 25 km of JFK, nearest first.
GET /tweets/_search
{
  "size": 100,
  "track_total_hits": true,
  "query": {
    "geo_distance": {
      "distance": "25km",
      "location": {"lat": 40.6413, "lon": -73.7781}
    }
  },
  "sort": [
    {"_geo_distance": {
      "location": {"lat": 40.6413, "lon": -73.7781},
      "order": "asc",
      "unit": "km",
      "distance_type": "arc"
    }},
    {"tweet_id": "asc"}
  ],
  "_source": ["text", "airline", "airline_sentiment", "location"]
}
